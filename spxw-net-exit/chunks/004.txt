function broker(){ return $('broker')?$('broker').value:'robinhood'; }
function defaultBrokerProfile(b=broker()){
  return b==='tradestation'?'ts_tier1':b==='tradier'?'tradier_pro':b==='tastytrade'?'tasty_standard':b==='schwab'?'schwab_online':'rh_gold';
}
function brokerProfile(){ return $('brokerProfile')?$('brokerProfile').value:defaultBrokerProfile(); }
function brokerLabel(){
  return broker()==='tradestation'?'TradeStation':broker()==='tradier'?'Tradier':broker()==='tastytrade'?'tastytrade':broker()==='schwab'?'Charles Schwab / thinkorswim':'Robinhood';
}
function profileLabel(){
  if(broker()==='tradestation') return (TS_TIERS[brokerProfile()]||TS_TIERS.ts_tier1).label;
  if(broker()==='tradier') return (TRADIER_PROFILES[brokerProfile()]||TRADIER_PROFILES.tradier_pro).label;
  if(broker()==='tastytrade') return (TASTY_PROFILES[brokerProfile()]||TASTY_PROFILES.tasty_standard).label;
  if(broker()==='schwab') return (SCHWAB_PROFILES[brokerProfile()]||SCHWAB_PROFILES.schwab_online).label;
  return (RH_PROFILES[brokerProfile()]||RH_PROFILES.rh_gold).label;
}
function brokerDisplayLabel(){
  if(broker()==='tradestation') return `TradeStation · ${(TS_TIERS[brokerProfile()]||TS_TIERS.ts_tier1).label}`;
  if(broker()==='tradier') return `Tradier · ${(TRADIER_PROFILES[brokerProfile()]||TRADIER_PROFILES.tradier_pro).shortLabel}`;
  if(broker()==='tastytrade') return `tastytrade · ${(TASTY_PROFILES[brokerProfile()]||TASTY_PROFILES.tasty_standard).shortLabel}`;
  if(broker()==='schwab') return `Schwab / thinkorswim · ${(SCHWAB_PROFILES[brokerProfile()]||SCHWAB_PROFILES.schwab_online).shortLabel}`;
  return `Robinhood · ${(RH_PROFILES[brokerProfile()]||RH_PROFILES.rh_gold).shortLabel}`;
}
function tsPerContractSide(profile=brokerProfile()){ const t=TS_TIERS[profile]||TS_TIERS.ts_tier1; return t.multiLeg+t.indexFee+t.occ; }
function rhPerContractSide(profile=brokerProfile(),premiumBand='mixed'){
  const p=RH_PROFILES[profile]||RH_PROFILES.rh_gold;
  const exchange=premiumBand==='high'?.59:premiumBand==='low'?.50:.545;
  return p.contractFee+exchange+.04;
}
function tradierPerContractSide(profile=brokerProfile()){
  const p=TRADIER_PROFILES[profile]||TRADIER_PROFILES.tradier_pro;
  // Published fee-schedule estimate for balanced multi-leg SPXW:
  // commission + SPX index fee + clearing + ORF + CAT + half-weighted sell-side TAF.
  return p.commission + .60 + .0775 + .0187 + .0072 + (.00279/2);
}
function tastyFeesPerContract(){
  const p=TASTY_PROFILES[brokerProfile()]||TASTY_PROFILES.tasty_standard;
  return {open:p.openCommission+p.spxFee,close:p.closeCommission+p.spxFee};
}
function schwabPerContractSide(){ return (SCHWAB_PROFILES[brokerProfile()]||SCHWAB_PROFILES.schwab_online).commission; }
function configureBrokerProfile(){
  const sel=$('brokerProfile'); if(!sel) return;
  const prev=sel.value;
  if(broker()==='tradestation'){
    sel.innerHTML=Object.entries(TS_TIERS).map(([k,v])=>`<option value="${k}">${v.label}</option>`).join('');
    sel.value=TS_TIERS[prev]?prev:'ts_tier1';
    $('brokerProfileHint').textContent='TradeStation: pricing por volumen mensual de options. Tier 1 es el estándar para 0–500 contratos/mes.';
  } else if(broker()==='tradier'){
    sel.innerHTML=Object.entries(TRADIER_PROFILES).map(([k,v])=>`<option value="${k}">${v.label}</option>`).join('');
    sel.value=TRADIER_PROFILES[prev]?prev:'tradier_pro';
    $('brokerProfileHint').textContent='Tradier: Lite/Pro usan $0.35 por contrato de single-listed index; Pro Plus $0.10. La suscripción mensual no se carga al trade.';
  } else if(broker()==='tastytrade'){
    sel.innerHTML=Object.entries(TASTY_PROFILES).map(([k,v])=>`<option value="${k}">${v.label}</option>`).join('');
    sel.value=TASTY_PROFILES[prev]?prev:'tasty_standard';
    $('brokerProfileHint').textContent='tastytrade: $1/contract para abrir broad-based index options y $0 para cerrar; SPX tiene fee de índice publicado de $0.65/contract. Otros fees aplican.';
  } else if(broker()==='schwab'){
    sel.innerHTML=Object.entries(SCHWAB_PROFILES).map(([k,v])=>`<option value="${k}">${v.label}</option>`).join('');
    sel.value=SCHWAB_PROFILES[prev]?prev:'schwab_online';
    $('brokerProfileHint').textContent='Charles Schwab / thinkorswim: $0.65 por contrato online. Industry fees (ORF/index fee) pueden agregarse y varían; usa fill real para precisión.';
  } else {
    sel.innerHTML=Object.entries(RH_PROFILES).map(([k,v])=>`<option value="${k}">${v.label}</option>`).join('');
    sel.value=RH_PROFILES[prev]?prev:'rh_gold';
    $('brokerProfileHint').textContent='Robinhood: Gold cobra $0.35/contrato de index options; Standard sin Gold cobra $0.50. SPXW exchange + ORF/OCC se calculan aparte.';
  }
}
function brokerAutoFeesPerContract(){
  if(broker()==='tradestation'){ const p=tsPerContractSide(); return {open:p,close:p}; }
  if(broker()==='tradier'){ const p=tradierPerContractSide(); return {open:p,close:p}; }
  if(broker()==='tastytrade') return tastyFeesPerContract();
  if(broker()==='schwab'){ const p=schwabPerContractSide(); return {open:p,close:p}; }
  const p=rhPerContractSide(brokerProfile(),'mixed'); return {open:p,close:p};
}
function brokerAutoPerContractSide(){ const f=brokerAutoFeesPerContract(); return (f.open+f.close)/2; }


function money(n){ if(!Number.isFinite(n)) return '—'; return (n<0?'-$':'$')+Math.abs(n).toFixed(2); }
function signedMoney(n){ if(!Number.isFinite(n)) return '—'; return (n>=0?'+':'-')+'$'+Math.abs(n).toFixed(2); }
function num(id){ return parseFloat($(id).value)||0; }
function qty(){ return Math.max(1,Math.floor(num('qty'))); }
function legs(){ return +$('strategy').value; }
function totalContractsSide(){ return qty()*legs(); }
function initialGrossCredit(){ return num('credit')*100*qty(); }
function targetDollarsFromPct(pct){ return initialGrossCredit()*(pct/100); }
function targetPctFromDollars(dollars){ const base=initialGrossCredit(); return base>0?dollars/base*100:0; }
function syncTargetFields(from='dollars'){
  if(from==='percent'){
    targetInputMode='percent';
    const dollars=targetDollarsFromPct(num('targetPct'));
    $('target').value=dollars.toFixed(2);
  } else {
    targetInputMode='dollars';
    const pct=targetPctFromDollars(num('target'));
    $('targetPct').value=pct.toFixed(1);
  }
}
