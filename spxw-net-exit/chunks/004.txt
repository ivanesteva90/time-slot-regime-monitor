    obs=[{id:Date.now(),date:'2026-08-28',desc:'2 SPXW credit spreads · closing fill',contracts:4,fee:3.74,broker:'robinhood',profile:'rh_gold'}];
  }
  // Migration: historical v4.2 observations were Robinhood fills and had no broker field.
  obs=obs.map(o=>({...o,broker:o.broker||'robinhood',profile:o.profile||(o.broker==='tradestation'?'ts_tier1':'rh_gold')}));

}

function observedPerContract(){
  const obs=currentBrokerObservations().filter(o=>o.contracts>0 && o.fee>=0);
  const totalContracts=obs.reduce((s,o)=>s+Number(o.contracts),0);
  const totalFees=obs.reduce((s,o)=>s+Number(o.fee),0);
  return totalContracts>0?totalFees/totalContracts:null;
}
