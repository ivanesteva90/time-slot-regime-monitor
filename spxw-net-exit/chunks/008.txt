function loadTrade(){
  const p=JSON.parse(localStorage.getItem(STORAGE_TRADE)||'null');
  if(!p){ $('savedNote').textContent='No hay un trade guardado todavía.'; return; }
  $('broker').value=p.broker||'robinhood'; configureBrokerProfile(); if(p.brokerProfile) $('brokerProfile').value=p.brokerProfile;
  ['strategy','position','qty','credit','current','target','targetPct','maxLoss','maxLossPct','maxHold','width1','width2','openFees','closeFees'].forEach(id=>{ if(p[id]!==undefined) $(id).value=p[id]; });

  lossInputMode=p.lossInputMode||'percent';

function clearTrade(){
  localStorage.removeItem(STORAGE_TRADE); localStorage.removeItem(STORAGE_TIMER); stopTimerLoop();
  $('savedNote').textContent='Trade guardado y timer eliminados.'; renderTimer();
}

function startTimer(){
  const mins=Math.max(1,num('maxHold'));
  localStorage.setItem(STORAGE_TIMER,JSON.stringify({start:Date.now(),durationMs:mins*60000}));
  startTimerLoop(); renderTimer();
}
function stopTimerLoop(){ if(timerInterval){clearInterval(timerInterval);timerInterval=null;} }
function startTimerLoop(){ stopTimerLoop(); timerInterval=setInterval(renderTimer,500); }
function renderTimer(){
  const p=JSON.parse(localStorage.getItem(STORAGE_TIMER)||'null');
  const maxHold=Math.max(1,num('maxHold'));
  if(!p){
    $('timerValue').textContent=`${String(maxHold).padStart(2,'0')}:00`;
    $('timerPill').className='statusPill'; $('timerPill').textContent='No iniciado'; $('timerBar').style.width='0%'; $('timerBox').classList.remove('expired'); return;
  }
  const elapsed=Date.now()-p.start, remain=p.durationMs-elapsed, pct=clamp(elapsed/p.durationMs*100,0,100);
  $('timerBar').style.width=pct.toFixed(1)+'%';
  if(remain<=0){
    $('timerValue').textContent='00:00'; $('timerPill').className='statusPill bad'; $('timerPill').textContent='TIME STOP'; $('timerBox').classList.add('expired'); stopTimerLoop();
  } else {
    const sec=Math.ceil(remain/1000), m=Math.floor(sec/60), s=sec%60;
    $('timerValue').textContent=`${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
    $('timerPill').className='statusPill good'; $('timerPill').textContent='Activo'; $('timerBox').classList.remove('expired');
  }
}

function renderObservations(){
  const obs=currentBrokerObservations();
  const avg=observedPerContract();
  $('feeAvgPill').textContent=avg!==null?`Avg ${brokerDisplayLabel()}: ${money(avg)}/contract/side`:`${brokerDisplayLabel()}: sin fills observados`;
  $('obsTable').innerHTML=obs.length?obs.slice().reverse().map(o=>`<tr><td>${o.date||'—'}</td><td>${escapeHtml(o.desc||'Fill')}</td><td>${o.contracts}</td><td>${money(Number(o.fee))}</td><td>${money(Number(o.fee)/Number(o.contracts))}</td><td><button class="btn small danger" data-del="${o.id}" type="button">Eliminar</button></td></tr>`).join(''):`<tr><td colspan="6" class="muted">Sin fills guardados para ${brokerDisplayLabel()}.</td></tr>`;
  document.querySelectorAll('[data-del]').forEach(b=>b.onclick=()=>{ const id=Number(b.dataset.del); localStorage.setItem(STORAGE_OBS,JSON.stringify(getObservations().filter(o=>o.id!==id))); renderObservations(); update(); });
}
function escapeHtml(s){ return String(s).replace(/[&<>'"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c])); }
function addObservation(){
  const contracts=Math.max(1,Math.floor(num('obsContracts'))), fee=num('obsFee');
  if(!(fee>=0)){return;}
  const obs=getObservations();
  obs.push({id:Date.now(),date:$('obsDate').value||todayISO(),desc:$('obsDesc').value.trim()||'SPXW fill',contracts,fee,broker:broker(),profile:brokerProfile()});
  localStorage.setItem(STORAGE_OBS,JSON.stringify(obs)); renderObservations(); update();
}
function clearObservations(){
  const keep=getObservations().filter(o=>!((o.broker||'robinhood')===broker() && (o.profile||(o.broker==='tradestation'?'ts_tier1':'rh_gold'))===brokerProfile()));
  localStorage.setItem(STORAGE_OBS,JSON.stringify(keep)); renderObservations(); update();
}
