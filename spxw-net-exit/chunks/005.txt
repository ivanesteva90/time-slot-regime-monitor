):tsPerContractSide();
    const side=perContract*contracts;
    return {open:side,close:side,source:feeMode,perContract};
  }

  // Robinhood SPXW current retail model.
  // Gold: $0.35 contract fee; Standard: $0.50 contract fee.
  // SPXW exchange: $0.50 when leg premium < $1; $0.59 when >= $1.
  // Combined ORF/OCC collected by Robinhood: $0.04/contract.
  const low=rhPerContractSide(brokerProfile(),'low'), high=rhPerContractSide(brokerProfile(),'high');
  let perStrategySide;
  if(feeMode==='conservative') perStrategySide=legs()*high;
  else {
    const hi=Math.floor(legs()/2), lo=legs()-hi;
    perStrategySide=hi*high+lo*low;
  }
  const side=perStrategySide*qty();
  return {open:side,close:side,source:feeMode,low,high};
}

function solveProfit(target){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit-(target+fees)/(100*q);
  if(raw<.05-1e-9) return {possible:false,raw,fees,f,maxNet:credit*100*q-fees};
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  const pct=credit>0?(credit-close)/credit*100:0;
  return {possible:true,close,gross,net,pct,fees,f,raw,maxNet:credit*100*q-fees};
}

function solveStop(maxLoss){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit+(maxLoss-fees)/(100*q);
  // Use the highest valid tick that does not exceed the selected net loss.
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  return {close,net,gross,raw,fees};
}

function solveStopMultiple(mult=1.50){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit*mult;
  // Round down so the executable tick does not exceed the selected gross multiple.
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  const grossLossPct=credit>0?Math.max(0,(close-credit)/credit*100):0;
  return {close,net,gross,raw,fees,mult,grossLossPct};
}

function activeStop(){
  return stopRuleMode==='multiple' ? solveStopMultiple(stopMultiple) : solveStop(num('maxLoss'));
}

function currentMetrics(){
  const credit=num('credit'), current=num('current'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const gross=(credit-current)*100*q;
  const net=gross-fees;
  const pct=credit>0?(credit-current)/credit*100:0;
  return {current,gross,net,pct,fees};
}
