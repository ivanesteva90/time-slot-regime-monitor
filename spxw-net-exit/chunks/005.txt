function observedPerContract(){
  const obs=currentBrokerObservations().filter(o=>o.contracts>0 && o.fee>=0);
  const totalContracts=obs.reduce((s,o)=>s+Number(o.contracts),0);
  const totalFees=obs.reduce((s,o)=>s+Number(o.fee),0);
  return totalContracts>0?totalFees/totalContracts:null;
}

function calcFees(){
  const contracts=totalContractsSide();
  if(feeMode==='manual') return {open:num('openFees'),close:num('closeFees'),source:'manual'};
  if(feeMode==='observed'){
    const avg=observedPerContract();
    const perContract=avg!==null?avg:brokerAutoPerContractSide();
    const side=perContract*contracts;
    return {open:side,close:side,source:avg!==null?'observed':'observed-fallback'};
  }

  if(broker()==='tradestation'){
    const perContract=feeMode==='conservative'?tsPerContractSide('ts_tier1'):tsPerContractSide();
    const side=perContract*contracts;
    return {open:side,close:side,source:feeMode,perContract};
  }
  if(broker()==='tradier'){
    const base=tradierPerContractSide();
    // Conservative adds a small 5-cent/contract allowance because pass-through regulatory rates can change.
    const perContract=feeMode==='conservative'?base+.05:base;
    return {open:perContract*contracts,close:perContract*contracts,source:feeMode,perContract};
  }
  if(broker()==='tastytrade'){
    const pc=tastyFeesPerContract();
    // Published base includes broad-index commission + SPX proprietary index fee.
    // Clearing/regulatory fees are broker/execution dependent; conservative adds a transparent $0.15/contract/side allowance.
    const buffer=feeMode==='conservative'?.15:0;
    return {open:(pc.open+buffer)*contracts,close:(pc.close+buffer)*contracts,source:feeMode,openPerContract:pc.open+buffer,closePerContract:pc.close+buffer};
  }
  if(broker()==='schwab'){
    const base=schwabPerContractSide();
    // Schwab publishes $0.65/contract and separately passes an Industry Fee that can include ORF and proprietary SPX fees.
    // Conservative uses a $0.65/contract allowance for those variable index/industry charges; use fill real for exact costs.
    const perContract=feeMode==='conservative'?base+.65:base;
    return {open:perContract*contracts,close:perContract*contracts,source:feeMode,perContract};
  }

  // Robinhood SPXW current retail model.
  // Gold: $0.35 contract fee; Standard: $0.50 contract fee.
  // SPXW exchange: $0.50 when leg premium < $1; $0.59 when >= $1.
  // Combined ORF/OCC collected by Robinhood: $0.04/contract.
  const low=rhPerContractSide(brokerProfile(),'low'), high=rhPerContractSide(brokerProfile(),'high');
  let perStrategySide;
  if(feeMode==='conservative') perStrategySide=legs()*high;
  else {
    const hi=Math.floor(legs()/2), lo=legs()-hi;
    perStrategySide=hi*high+lo*low;
  }
  const side=perStrategySide*qty();
  return {open:side,close:side,source:feeMode,low,high};
}

function solveProfit(target){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit-(target+fees)/(100*q);
  if(raw<.05-1e-9) return {possible:false,raw,fees,f,maxNet:credit*100*q-fees};
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  const pct=credit>0?(credit-close)/credit*100:0;
  return {possible:true,close,gross,net,pct,fees,f,raw,maxNet:credit*100*q-fees};
}

function solveStop(maxLoss){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit+(maxLoss-fees)/(100*q);
  // Use the highest valid tick that does not exceed the selected net loss.
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  return {close,net,gross,raw,fees};
}

function solveStopMultiple(mult=1.50){
  const credit=num('credit'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const raw=credit*mult;
  // Round down so the executable tick does not exceed the selected gross multiple.
  const close=round2(Math.max(.05,tickFloor(raw,.05)));
  const gross=(credit-close)*100*q;
  const net=gross-fees;
  const grossLossPct=credit>0?Math.max(0,(close-credit)/credit*100):0;
  return {close,net,gross,raw,fees,mult,grossLossPct};
}

function activeStop(){
  return stopRuleMode==='multiple' ? solveStopMultiple(stopMultiple) : solveStop(num('maxLoss'));
}

function currentMetrics(){
  const credit=num('credit'), current=num('current'), q=qty(), f=calcFees(), fees=f.open+f.close;
  const gross=(credit-current)*100*q;
  const net=gross-fees;
  const pct=credit>0?(credit-current)/credit*100:0;
  return {current,gross,net,pct,fees};
}
