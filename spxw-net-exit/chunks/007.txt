)}× crédito (bruto)`;
    $('stopMeta').textContent=`${stopMultiple.toFixed(2)}× de $${credit.toFixed(2)} = $${stop.raw.toFixed(3)} raw → tick $${stop.close.toFixed(2)}. Pérdida bruta: −${stop.grossLossPct.toFixed(1)}%; después de fees: ${signedMoney(stop.net)} = −${stopNetPct.toFixed(1)}% NET.`;
  } else {
    stopEyebrow.textContent='Stop sugerido · pérdida NET';
    $('stopMeta').textContent=`Target −${maxLossPct.toFixed(1)}% NET (${money(maxLoss)}). En $${stop.close.toFixed(2)}: ${signedMoney(stop.net)} = −${stopNetPct.toFixed(1)}% NET. Raw matemático: $${stop.raw.toFixed(3)}.`;
  }
  if(current>=stop.close-1e-9){ $('stopMeta').textContent += '  ⚠ El precio actual ya alcanzó/superó este umbral.'; }

  if(r.possible){
    const reward=Math.max(0,r.net);
    const risk=Math.max(0,-stop.net);
    const rewardRisk=risk>0?reward/risk:Infinity;
    const winsPerLoss=reward>0?risk/reward:Infinity;
    const beWR=(reward+risk)>0?risk/(reward+risk)*100:0;
    $('rewardRisk').textContent=Number.isFinite(rewardRisk)?rewardRisk.toFixed(2):'∞';
    $('winsPerLoss').textContent=Number.isFinite(winsPerLoss)?winsPerLoss.toFixed(2)+'×':'∞';
    $('breakEvenWR').textContent=beWR.toFixed(1)+'%';
    $('planLabel').textContent=stopRuleMode==='multiple'
      ? `TP +${targetPct.toFixed(1)}% NET / SL ${stopMultiple.toFixed(2)}× crédito · real +${netPct(r.net).toFixed(1)}% / −${stopNetPct.toFixed(1)}% NET`
      : `TP +${targetPct.toFixed(1)}% / SL −${maxLossPct.toFixed(1)}% NET · real +${netPct(r.net).toFixed(1)}% / −${stopNetPct.toFixed(1)}%`;
    const isBase=stopRuleMode==='net' && Math.abs(targetPct-30)<.05 && Math.abs(maxLossPct-50)<.05;
    const isFast=stopRuleMode==='net' && Math.abs(targetPct-25)<.05 && Math.abs(maxLossPct-50)<.05;
    const is15=stopRuleMode==='multiple' && Math.abs(stopMultiple-1.5)<.001;
    $('planPill').className='statusPill '+(isBase?'good':isFast||is15?'warn':'');
    $('planPill').textContent=isBase?'BASE 30/50':isFast?'FAST 25/50':is15?'1.5× SL':'CUSTOM';
    $('planNote').textContent=`Con los ticks y fees actuales, necesitas ~${beWR.toFixed(1)}% de winners para break-even si cada winner/loss terminara exactamente en estos exits. El modelo no asume que ese win rate vaya a ocurrir.`;
  } else {
    $('rewardRisk').textContent='—'; $('winsPerLoss').textContent='—'; $('breakEvenWR').textContent='—'; $('planLabel').textContent='Target de profit no alcanzable con cierre activo ≥ $0.05'; $('planPill').className='statusPill bad'; $('planPill').textContent='REVISAR';
  }

  const width=isIC?Math.max(num('width1'),num('width2')):num('width1');
  const maxGross=credit*100*q;
  const maxNetAtZero=maxGross-totalFees;
  const maxLossTheory=width>0?(width-credit)*100*q+totalFees:null;
  const rr=maxLossTheory&&target>0?maxLossTheory/target:null;
  $('detail').innerHTML=`
    <div class="rowline"><span>Broker / pricing</span><b>${brokerLabel()} · ${profileLabel()}</b></div>
    <div class="rowline"><span>Crédito bruto inicial</span><b>${money(maxGross)}</b></div>
    <div class="rowline"><span>Objetivo NETO</span><b>${money(target)} · ${targetPct.toFixed(1)}%</b></div>
    <div class="rowline"><span>Stop seleccionado</span><b>${stopRuleMode==='multiple'?`${stopMultiple.toFixed(2)}× crédito · ${signedMoney(stop.net)} (${netPct(stop.net).toFixed(1)}% NET)`:`−${money(maxLoss).replace('$','')} · −${maxLossPct.toFixed(1)}% NET`}</b></div>
    <div class="rowline"><span>Fees apertura</span><b>${money(f.open)}</b></div>
    <div class="rowline"><span>Fees cierre</span><b>${money(f.close)}</b></div>
    <div class="rowline"><span>Máx. neto teórico a $0.00*</span><b>${money(maxNetAtZero)}</b></div>
    ${maxLossTheory!==null?`<div class="rowline"><span>Máx. pérdida teórica aprox.</span><b>${money(maxLossTheory)}</b></div>`:''}
    ${rr?`<div class="rowline"><span>Máx. riesgo / target neto</span><b>${rr.toFixed(1)}×</b></div>`:''}
  `;

  updateSelectionVisuals();

  $('targetTable').innerHTML=quickTargets.map(t=>{
    const x=solveProfit(t), tPct=targetPctFromDollars(t);
    if(!x.possible) return `<tr><td><b>$${t}</b></td><td>${tPct.toFixed(1)}%</td><td colspan="6">Requiere cierre &lt; $0.05</td></tr>`;
    const cls=(Math.abs(t-target)<.001)?' class="hit"':'';
    return `<tr${cls}><td><b>$${t}</b></td><td>${tPct.toFixed(1)}%</td><td><b>$${x.close.toFixed(2)}</b></td><td>${x.pct.toFixed(1)}%</td><td>${money(x.gross)}</td><td>${money(x.fees)}</td><td>${money(x.net)}</td><td><b>${netPct(x.net).toFixed(1)}%</b></td></tr>`;
  }).join('');

  const planRows=[
    {tp:25,sl:50,type:'net'},
    {tp:30,sl:50,type:'net'},
    {tp:30,mult:1.5,type:'multiple'},
    {tp:35,sl:50,type:'net'},
    {tp:30,sl:40,type:'net'},
    {tp:30,sl:60,type:'net'}
  ];
  $('riskPlanTable').innerHTML=planRows.map(p=>{
    const tpD=targetDollarsFromPct(p.tp), px=solveProfit(tpD);
    const sx=p.type==='multiple'?solveStopMultiple(p.mult):solveStop(lossDollarsFromPct(p.sl));
    const label=p.type==='multiple'?`${p.tp}% / ${p.mult.toFixed(1)}×`:`${p.tp}% / ${p.sl}%`;
    if(!px.possible) return `<tr><td><b>${label}</b></td><td colspan="6">TP no alcanzable con cierre ≥ $0.05</td></tr>`;
    const reward=Math.max(0,px.net), risk=Math.max(0,-sx.net);
    const rr=risk>0?reward/risk:Infinity, be=(reward+risk)>0?risk/(reward+risk)*100:0;
    const cls=(p.type==='net'&&p.tp===30&&p.sl===50)?' class="hit"':'';
    const suffix=p.type==='net'&&p.tp===30&&p.sl===50?' · BASE':p.type==='multiple'?' · GROSS':'';
    return `<tr${cls}><td><b>${label}${suffix}</b></td><td>$${px.close.toFixed(2)}</td><td>+${netPct(px.net).toFixed(1)}%</td><td>$${sx.close.toFixed(2)}</td><td>−${Math.abs(netPct(sx.net)).toFixed(1)}%</td><td>${Number.isFinite(rr)?rr.toFixed(2):'∞'}</td><td><b>${be.toFixed(1)}%</b></td></tr>`;
  }).join('');

  updateFeeDesc();
}
