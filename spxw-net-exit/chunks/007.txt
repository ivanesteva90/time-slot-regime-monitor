/contract/side de comisión online. Industry Fee (ORF + proprietary SPX fee, según ejecución) no incluido en Auto.`
      : feeMode==='conservative'?`Schwab conservador: $0.65 commission + $0.65/contract de allowance para Industry/index fees. No es un fee fijo; usa confirmation real para exactitud.`:`Manual: usa exactamente fees de apertura y cierre del confirmation de Schwab.`;
  } else {
    const p=RH_PROFILES[brokerProfile()]||RH_PROFILES.rh_gold; const low=rhPerContractSide(brokerProfile(),'low'), high=rhPerContractSide(brokerProfile(),'high');
    $('feeDesc').textContent = feeMode==='mixed'
      ? `${p.shortLabel} Robinhood: ${money(low)}/contract/side si la leg SPXW tiene premium < $1 y ${money(high)} si ≥ $1. Auto mixto aproxima mitad low / mitad high.`
      : feeMode==='conservative'?`${p.shortLabel} Robinhood conservador: asume todas las legs a ${money(high)}/contract/side (premium ≥ $1).`:`Manual: usa exactamente los fees reales de Robinhood (${p.shortLabel}).`;
  }
}

function setMode(m){
  feeMode=m;
  ['observedBtn','autoBtn','conBtn','manualBtn'].forEach(id=>$(id).classList.remove('active'));
  if(m==='observed')$('observedBtn').classList.add('active');
  if(m==='mixed')$('autoBtn').classList.add('active');
  if(m==='conservative')$('conBtn').classList.add('active');
  if(m==='manual')$('manualBtn').classList.add('active');
  $('manualFees').classList.toggle('show',m==='manual');
  updateBrokerUI(); updateFeeDesc(); update();
}
