  if($('conBtn')) $('conBtn').textContent=broker()==='tradestation'?'Auto · Tier 1 max':'Auto · conservador';

function updateFeeDesc(){
  const avg=observedPerContract();
  const n=currentBrokerObservations().length;
  if(feeMode==='observed'){
    $('feeDesc').textContent=avg!==null
      ? `Observed history (${brokerDisplayLabel()}): usa ${money(avg)} por contrato por lado según ${n} fill(s) guardado(s).`
      : `Observed history (${brokerDisplayLabel()}): todavía no hay fills guardados; temporalmente usa el estimado Auto del perfil ${profileLabel()}.`;
    return;
  }
  if(broker()==='tradestation'){
    const t=TS_TIERS[brokerProfile()]||TS_TIERS.ts_tier1;
    const pc=tsPerContractSide();
    $('feeDesc').textContent=feeMode==='mixed'
      ? `${t.label}: $${t.multiLeg.toFixed(2)} multi-leg + $${t.indexFee.toFixed(2)} index fee + $${t.occ.toFixed(3)} OCC = ${money(pc)}/contract/side. ORF y cargos regulatorios variables no están incluidos en Auto.`
      : feeMode==='conservative'
      ? `Conservador TradeStation: usa Tier 1 publicado ($1.425/contract/side) aunque tengas un tier mejor. Aun así, ORF/regulatorios variables pueden aparecer en el fill.`
      : `Manual: usa exactamente los fees reales de apertura y cierre que te muestre TradeStation.`;
  } else {
    const p=RH_PROFILES[brokerProfile()]||RH_PROFILES.rh_gold;
    const low=rhPerContractSide(brokerProfile(),'low'), high=rhPerContractSide(brokerProfile(),'high');
    $('feeDesc').textContent = feeMode==='mixed'
      ? `${p.shortLabel} Robinhood: ${money(low)}/contract/side si la leg SPXW tiene premium < $1 y ${money(high)} si ≥ $1. Auto mixto aproxima mitad low / mitad high.`
      : feeMode==='conservative'
      ? `${p.shortLabel} Robinhood conservador: asume todas las legs a ${money(high)}/contract/side (premium ≥ $1).`
      : `Manual: usa exactamente los fees reales de apertura y cierre que te muestre Robinhood (${p.shortLabel}).`;
  }
}

function setMode(m){
  feeMode=m;
  ['observedBtn','autoBtn','conBtn','manualBtn'].forEach(id=>$(id).classList.remove('active'));
  if(m==='observed')$('observedBtn').classList.add('active');
  if(m==='mixed')$('autoBtn').classList.add('active');
  if(m==='conservative')$('conBtn').classList.add('active');
  if(m==='manual')$('manualBtn').classList.add('active');
  $('manualFees').classList.toggle('show',m==='manual');
  updateBrokerUI(); updateFeeDesc(); update();
}
