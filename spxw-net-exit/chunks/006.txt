ick=()=>{ $('targetPct').value=b.dataset.pctTarget; syncTargetFields('percent'); update(); });
  document.querySelectorAll('[data-loss-pct]').forEach(b=>b.onclick=()=>{ stopRuleMode='net'; $('maxLossPct').value=b.dataset.lossPct; syncLossFields('percent'); update(); });
  updateSelectionVisuals();
}
function setSelected(id,on){
  const el=$(id); if(!el) return;
  el.classList.toggle('selected',!!on);
  el.setAttribute('aria-pressed',on?'true':'false');
}
function updateSelectionVisuals(){
  const tpPct=num('targetPct'), tpDollar=num('target'), slPct=num('maxLossPct');
  const isNet=stopRuleMode==='net';
  const is15=stopRuleMode==='multiple' && Math.abs(stopMultiple-1.5)<.001;
  setSelected('apply3050',isNet && Math.abs(tpPct-30)<.05 && Math.abs(slPct-50)<.05);
  setSelected('apply2550',isNet && Math.abs(tpPct-25)<.05 && Math.abs(slPct-50)<.05);
  setSelected('apply3015',is15 && Math.abs(tpPct-30)<.05);
  setSelected('apply15stop',is15);
  document.querySelectorAll('[data-dollar-target]').forEach(b=>{
    const on=Math.abs(Number(b.dataset.dollarTarget)-tpDollar)<.01;
    b.classList.toggle('selected',on); b.setAttribute('aria-pressed',on?'true':'false');
  });
  document.querySelectorAll('[data-pct-target]').forEach(b=>{
    const on=Math.abs(Number(b.dataset.pctTarget)-tpPct)<.05;
    b.classList.toggle('selected',on); b.setAttribute('aria-pressed',on?'true':'false');
  });
  document.querySelectorAll('[data-loss-pct]').forEach(b=>{
    const on=isNet && Math.abs(Number(b.dataset.lossPct)-slPct)<.05;
    b.classList.toggle('selected',on); b.setAttribute('aria-pressed',on?'true':'false');
  });
}

function applyRiskPlan(tpPct,slPct){
  stopRuleMode='net';
  $('targetPct').value=Number(tpPct).toFixed(1);
  $('maxLossPct').value=Number(slPct).toFixed(1);
  syncTargetFields('percent'); syncLossFields('percent'); update();
}
function applyMultipleStop(mult=1.50,tpPct=null){
  stopRuleMode='multiple'; stopMultiple=Number(mult)||1.50;
  if(tpPct!==null){ $('targetPct').value=Number(tpPct).toFixed(1); syncTargetFields('percent'); }
  update();
}
