  $('timerValue').textContent=`${String(maxHold).padStart(2,'0')}:00`;
    $('timerPill').className='statusPill'; $('timerPill').textContent='No iniciado'; $('timerBar').style.width='0%'; $('timerBox').classList.remove('expired'); return;
  }
  const elapsed=Date.now()-p.start, remain=p.durationMs-elapsed, pct=clamp(elapsed/p.durationMs*100,0,100);
  $('timerBar').style.width=pct.toFixed(1)+'%';
  if(remain<=0){
    $('timerValue').textContent='00:00'; $('timerPill').className='statusPill bad'; $('timerPill').textContent='TIME STOP'; $('timerBox').classList.add('expired'); stopTimerLoop();
  } else {
    const sec=Math.ceil(remain/1000), m=Math.floor(sec/60), s=sec%60;
    $('timerValue').textContent=`${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
    $('timerPill').className='statusPill good'; $('timerPill').textContent='Activo'; $('timerBox').classList.remove('expired');
  }
}
